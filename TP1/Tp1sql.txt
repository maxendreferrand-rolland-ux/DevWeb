1)
CREATE DATABASE IF NOT EXISTS mezabi DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
2)
CREATE TABLE if not exists `a_categories` ( `CODE_CATEGORIE` INT(11) NOT NULL AUTO_INCREMENT, `DESIGNATION` VARCHAR(30) NOT NULL, Primary key (CODE_CATEGORIE) ) ENGINE=innoDB DEFAULT CHARSET=utf8;
3) 
CREATE TABLE if not exists `a_couleurs` ( `CODE_COULEURS` INT(11) NOT NULL AUTO_INCREMENT, `DESIGNATION` VARCHAR(30) NOT NULL, Primary key (CODE_COULEURS) ) ENGINE=innoDB DEFAULT CHARSET=utf8;
4)
CREATE TABLE if not exists `a_tailles` ( `CODE_TAILLE` INT(11) NOT NULL AUTO_INCREMENT, `DESIGNATION` VARCHAR(30) NOT NULL, Primary key (CODE_TAILLE) ) ENGINE=innoDB DEFAULT CHARSET=utf8;

5)
CREATE TABLE if not exists `articles` (
    `ID_ARTICLE` INT(11) NOT NULL AUTO_INCREMENT,
    `CODE_ARTICLE` VARCHAR(15) NOT NULL,
    `DESIGNATION` VARCHAR(30) NOT NULL,
    `CATEGORIES` INT(11) NOT NULL,
    Primary key (ID_ARTICLE),
    CONSTRAINT fk_categories FOREIGN KEY (CATEGORIES) REFERENCES a_categories(CODE_CATEGORIE)
    ) ENGINE=innoDB DEFAULT CHARSET=utf8;
6)
CREATE TABLE if not exists `stockprix` (
    `ARTICLE` INT(11) NOT NULL,
	`COULEUR` INT(11) NOT NULL,
    `TAILLE` INT(11) NOT NULL,
    `PRIX` DECIMAL(10,2) NOT NULL,
    `CODE_BARRE` VARCHAR(13) NOT NULL,
    `STOCK` INT(11) NOT NULL,
    CONSTRAINT fk_articles FOREIGN KEY (ARTICLE) REFERENCES articles(ID_ARTICLE),
	CONSTRAINT fk_couleur FOREIGN KEY (COULEUR) REFERENCES a_couleurs(CODE_COULEURS),
	CONSTRAINT fk_taille FOREIGN KEY (TAILLE) REFERENCES a_tailles(CODE_TAILLE),    
    PRIMARY KEY (`ARTICLE`, `COULEUR`, `TAILLE`)
    ) ENGINE=innoDB DEFAULT CHARSET=utf8;
